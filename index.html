1 建立 Google 試算表與開啟 Apps Script

先至 Google 雲端硬碟建立一個新的試算表 ➔ 點擊上方選單「擴充功能」➔ 選擇「Apps Script」。

2 貼上後台程式碼並「部署為 Web 應用程式」

將下方程式碼覆蓋貼上至 Apps Script，點擊「部署」➔「新增部署」➔「Web 應用程式」。
※ 關鍵設定：誰有存取權請選擇「所有人 (Anyone)」！ 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // 如果是第一列，自動建立表頭
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "填報時間", "姓名", "病歷號", "出生日期", "性別", "年齡", "評估階段",
        "目前體重", "身高", "BMI", "A1得分", "A2得分", "A3得分", "A4得分",
        "Sec B得分", "Sec C得分", "Sec D得分", "PG-SGA總分", "SGA分級",
        "熱量達標%", "蛋白質達標%", "症狀清單", "處方建議"
      ]);
    }
    
    sheet.appendRow([
      new Date().toLocaleString("zh-TW", { timeZone: "Asia/Taipei" }),
      data.patientName,
      data.patientId,
      data.patientDob,
      data.patientGender,
      data.patientAge,
      data.timePoint,
      data.currentWeight,
      data.height,
      data.bmi,
      data.scoreA1,
      data.scoreA2,
      data.scoreA3,
      data.scoreA4,
      data.scoreB,
      data.scoreC,
      data.scoreD,
      data.totalScore,
      data.sgaCategoryText,
      data.calorieTargetPct,
      data.proteinTargetPct,
      data.symptomsList,
      data.recommendationText
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "result": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
